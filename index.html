        // Base64 转 ArrayBuffer 工具

        function base64ToArrayBuffer(base64) {

            const binaryString = window.atob(base64);

            const len = binaryString.length;

            const bytes = new Uint8Array(len);

            for (let i = 0; i < len; i++) {

                bytes[i] = binaryString.charCodeAt(i);

            }

            return bytes.buffer;

        }


        // PCM16 转 WAV 音频流（用于神经网络真人语音合成播放）

        function pcm16ToWavBlob(pcmBuffer, sampleRate = 24000) {

            const numChannels = 1;

            const bitsPerSample = 16;

            const byteRate = (sampleRate * numChannels * bitsPerSample) / 8;

            const blockAlign = (numChannels * bitsPerSample) / 8;

            const dataSize = pcmBuffer.byteLength;

            const buffer = new ArrayBuffer(44 + dataSize);

            const view = new DataView(buffer);


            const writeStr = (offset, str) => {

                for (let i = 0; i < str.length; i++) {

                    view.setUint8(offset + i, str.charCodeAt(i));

                }

            };


            writeStr(0, 'RIFF');

            view.setUint32(4, 36 + dataSize, true);

            writeStr(8, 'WAVE');

            writeStr(12, 'fmt ');

            view.setUint32(16, 16, true);

            view.setUint16(20, 1, true);

            view.setUint16(22, numChannels, true);

            view.setUint32(24, sampleRate, true);

            view.setUint32(28, byteRate, true);

            view.setUint16(32, blockAlign, true);

            view.setUint16(34, bitsPerSample, true);

            writeStr(36, 'data');

            view.setUint32(40, dataSize, true);


            new Uint8Array(buffer, 44).set(new Uint8Array(pcmBuffer));

            return new Blob([buffer], { type: 'audio/wav' });

        }


        // 智能语速与自然呼吸停顿分析器

        function analyzeSpeechCadence(text, profileKey) {

            const profile = VoiceProfiles[profileKey] || VoiceProfiles.sweet;

            let rate = profile.rate;

            let pitch = profile.pitch;

            let pauseMs = 180;


            if (/[！!]{1,}|太棒|哇|哈哈|太开心|终于/.test(text)) {

                rate *= 1.06;

                pitch *= 1.04;

                pauseMs = 150;

            } else if (/[？?]$/.test(text)) {

                pitch *= 1.05;

                pauseMs = 220;

            } else if (/……|\.\.\.|呵欠|哈欠|困|失落|孤单|呜/.test(text)) {

                rate *= 0.88;

                pitch *= 0.94;

                pauseMs = 340;

            } else if (text.length > 26) {

                rate *= 1.04;

                pauseMs = 140;

            }

            return { rate, pitch, pauseMs };

        }

        }

            const text = (rawText || '').trim();

            if (!text) return;

            markUserActivity(true);


            // 若尚未唤醒或点睛，先完成阶段引导

            if (AppState.phase === 'WAITING_WAKE' || AppState.phase === 'BOOTING') {

                triggerWakeUp();

                return;

            }

            if (AppState.phase === 'AWAKENED_NO_EYES') {

                triggerDianjing();

                return;

            }


            pushSpeechBubble(text, true);

            setVisualCompanionMode('thinking');

            setExpression('leftUp');


            // 技能指令识别：变换颜色 / 变身

            if (/变色|变换颜色|换个颜色|换装|变身|换皮肤/.test(text)) {

                await new Promise(r => setTimeout(r, 500));

                triggerColorTransformSkill(true);

                return;

            }


            const startTime = performance.now();

            const replyObj = await generateSmartReply(text);

            const elapsed = performance.now() - startTime;

            // 保持至少 0.5 秒自然思考停顿，且总响应不超过 1 秒

            if (elapsed < 500) {

                await new Promise(r => setTimeout(r, 500 - elapsed));

            }


            setExpression(replyObj.emotion || 'normal');

            triggerSingleBounce();

            playOrderedDialogueSequence([replyObj.reply], () => {

                setVisualCompanionMode('listening');

            });

        }

        // 全双工高容错语音识别引擎（“灵犀灵犀”拥有最高抢断权重）

        function initVoiceRecognition() {

            const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

            if (!SpeechRecognition) {

                statusText.textContent = '当前浏览器不支持语音，可使用下方输入框对话';

                return;

            }

            if (AppState.recognition) {

                try { AppState.recognition.abort(); } catch (e) {}

            }


            const rec = new SpeechRecognition();

            rec.lang = 'zh-CN';

            rec.continuous = true;

            rec.interimResults = true;

            AppState.recognition = rec;


            rec.onstart = () => {

                AppState.isListeningActive = true;

                micToggleBtn.classList.add('bg-emerald-500/40', 'ring-2', 'ring-emerald-300');

            };


            rec.onresult = (event) => {

                let interim = '';

                let finalStr = '';

                for (let i = event.resultIndex; i < event.results.length; i++) {

                    const res = event.results[i];

                    if (res.isFinal) {

                        finalStr += res[0].transcript;

                    } else {

                        interim += res[0].transcript;

                    }

                }


                const heard = (finalStr || interim).trim();

                if (!heard) return;


                liveTranscriptPill.style.opacity = '1';

                liveTranscriptText.textContent = `听到：${heard}`;

                if (AppState.phase === 'ACTIVATED' && !AppState.isSpeaking) {

                    setVisualCompanionMode('listening');

                }


                // 最高权重唤醒检测：即使正在说话/思考也能秒熔断抢入新对话

                if (hasWakeWord(heard)) {

                    liveTranscriptText.textContent = `✨ 唤醒成功：${heard}`;

                    const restQuery = stripWakeWord(heard);


                    if (AppState.phase === 'WAITING_WAKE' || AppState.phase === 'BOOTING') {

                        triggerWakeUp();

                        return;

                    }

                    if (AppState.phase === 'AWAKENED_NO_EYES') {

                        triggerDianjing();

                        return;

                    }


                    if (finalStr) {

                        stopAllSpeech();

                        markUserActivity(true);

                        if (restQuery.length >= 2) {

                            handleUserConversation(restQuery);

                        } else {

                            setExpression('surprised');

                            triggerSingleBounce();

                            playOrderedDialogueSequence(['在呢在呢！灵犀听到啦，请吩咐~'], () => {

                                setExpression('normal');

                                setVisualCompanionMode('listening');

                            });

                        }

                    }

                    return;

                }


                // 已激活状态下，用户说出完整一句话自动进入自由对话

                if (finalStr && AppState.phase === 'ACTIVATED' && !AppState.isSpeaking) {

                    handleUserConversation(finalStr);

                }

            };


            rec.onerror = (e) => {

                if (e.error === 'not-allowed') {

                    AppState.isListeningActive = false;

                    statusText.textContent = '麦克风未授权，可点击屏幕或使用下方输入栏';

                }

            };


            rec.onend = () => {

                if (AppState.isListeningActive) {

                    setTimeout(() => {

                        try { rec.start(); } catch (e) {}

                    }, 180);

                }

            };


            try {

                rec.start();

            } catch (e) {}

        }

        // 60fps 主动画循环：自然呼吸感 + 随机眨眼 + 全向眼球跟随 + 16s打呵欠 + 30s物理惯性游走

        let lastFrameTime = performance.now();

        function animateLoop(now = performance.now()) {

            const dt = Math.min(0.05, (now - lastFrameTime) / 1000);

            lastFrameTime = now;

            const elapsedSec = now / 1000;


            // 1. 矢量表情形变插值

            EyeMorphEngine.update(dt);


            // 2. 自然随机眨眼（仅在点睛激活且非手动锁定表情时触发）

            if (AppState.hasEyes && !AppState.manualExpressionOverride) {

                if (!AppState.isBlinking && elapsedSec >= AppState.nextBlinkTime) {

                    AppState.isBlinking = true;

                    EyeMorphEngine.setTarget('blink');

                    setTimeout(() => {

                        AppState.isBlinking = false;

                        EyeMorphEngine.setTarget(AppState.expression);

                        AppState.nextBlinkTime = (performance.now() / 1000) + 2.6 + Math.random() * 3.2;

                    }, 145);

                }

            }


            // 3. 全向眼球跟随（上下左右平滑椭圆域跟随）

            const targetGazeX = Math.max(-26, Math.min(26, AppState.pointerNormX * 26));

            const targetGazeY = Math.max(-22, Math.min(22, AppState.pointerNormY * 22));

            AppState.currentGazeX += (targetGazeX - AppState.currentGazeX) * Math.min(1, dt * 10);

            AppState.currentGazeY += (targetGazeY - AppState.currentGazeY) * Math.min(1, dt * 10);

            eyesGazeLayer.setAttribute('transform', `translate(${AppState.currentGazeX.toFixed(2)}, ${AppState.currentGazeY.toFixed(2)})`);


            // 4. 优雅缓慢呼吸感 + 弹性回弹（零高频抖动）

            bodyScale.current += (bodyScale.target - bodyScale.current) * Math.min(1, dt * 12);

            const slowBreathe = 1 + Math.sin(elapsedSec * 1.65) * 0.022;

            const speakPulse = AppState.isSpeaking ? (1 + Math.sin(elapsedSec * 5.2) * 0.015) : 1;

            const totalScale = bodyScale.current * slowBreathe * speakPulse;

            bodyGroup.setAttribute('transform', `scale(${totalScale.toFixed(4)})`);


            // 5. 闲置状态机（16秒打呵欠/失落，30秒物理惯性游走）

            if (AppState.phase === 'ACTIVATED' && !AppState.isSpeaking) {

                const idleSec = (Date.now() - AppState.lastInteractionTime) / 1000;

                if (idleSec >= 16 && AppState.idleStage === 0) {

                    AppState.idleStage = 1;

                    triggerIdleYawnOrSad();

                } else if (idleSec >= 30 && AppState.idleStage === 1) {

                    AppState.idleStage = 2;

                    RoamPhysics.active = true;

                    pickNextRoamTarget();

                }

            }


            // 6. 物理惯性游走更新（先快后慢 Ease-Out + 加速惯性侧倾）

            if (RoamPhysics.active || RoamPhysics.progress < 1) {

                if (RoamPhysics.progress < 1) {

                    RoamPhysics.progress = Math.min(1, RoamPhysics.progress + dt / RoamPhysics.duration);

                    // 三次方先快后慢缓动曲线 (Ease-Out Cubic)

                    const t = RoamPhysics.progress;

                    const easeOut = 1 - Math.pow(1 - t, 3);

                    const prevX = RoamPhysics.x;

                    RoamPhysics.x = RoamPhysics.startX + (RoamPhysics.targetX - RoamPhysics.startX) * easeOut;

                    RoamPhysics.y = RoamPhysics.startY + (RoamPhysics.targetY - RoamPhysics.startY) * easeOut;


                    // 瞬时横向速度产生的物理惯性倾斜角

                    const vx = (RoamPhysics.x - prevX) / Math.max(0.008, dt);

                    const targetTilt = Math.max(-11, Math.min(11, -vx * 0.032));

                    RoamPhysics.tiltAngle += (targetTilt - RoamPhysics.tiltAngle) * Math.min(1, dt * 10);

                } else {

                    RoamPhysics.tiltAngle += (0 - RoamPhysics.tiltAngle) * Math.min(1, dt * 8);

                    if (RoamPhysics.active) {

                        RoamPhysics.nextMoveTimer -= dt;

                        if (RoamPhysics.nextMoveTimer <= 0) {

                            pickNextRoamTarget();

                        }

                    }

                }

                lingxiWrapper.style.transform = `translate3d(${RoamPhysics.x.toFixed(1)}px, ${RoamPhysics.y.toFixed(1)}px, 0) skewX(${RoamPhysics.tiltAngle.toFixed(2)}deg) rotate(${(-RoamPhysics.tiltAngle * 0.45).toFixed(2)}deg)`;

            } else if (lingxiWrapper.classList.contains('awakened')) {

                lingxiWrapper.style.transform = 'translate3d(0px, 0px, 0)';

            }


            requestAnimationFrame(animateLoop);

        }

            animateLoop();

            startBootSequence();

        };
